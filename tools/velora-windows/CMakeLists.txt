cmake_minimum_required(VERSION 3.22)
project(velora-windows LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPR REQUIRED IMPORTED_TARGET hyprland=0.57.0)
pkg_check_modules(LIBS REQUIRED IMPORTED_TARGET jsoncpp lua wayland-server pixman-1 libeis-1.0 xcb cairo)
add_library(velora-windows MODULE plugin.cpp canvas.cpp caption.cpp frame-preference.cpp)
target_compile_features(velora-windows PRIVATE cxx_std_23)
# Plugin reload must release the old inode when the updater replaces the module.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(velora-windows PRIVATE -fno-gnu-unique)
endif()
target_compile_definitions(velora-windows PRIVATE WLR_USE_UNSTABLE)
target_link_libraries(velora-windows PRIVATE PkgConfig::HYPR PkgConfig::LIBS)
install(TARGETS velora-windows LIBRARY DESTINATION share/velora/hypr)

cmake_minimum_required(VERSION 3.22)
project(velora-file-dialog VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
find_package(KF6KIO REQUIRED)
add_library(velora-file-dialog MODULE plugin.cpp)
target_link_libraries(velora-file-dialog PRIVATE Qt6::Widgets KF6::KIOFileWidgets)
install(TARGETS velora-file-dialog LIBRARY DESTINATION generic)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(file-dialog-test test.cpp)
    target_link_libraries(file-dialog-test PRIVATE Qt6::Test Qt6::Widgets KF6::KIOFileWidgets)
    add_test(NAME file-dialog COMMAND file-dialog-test)
    set_tests_properties(file-dialog PROPERTIES ENVIRONMENT "QT_QPA_GENERIC_PLUGINS=velora-file-dialog;QT_PLUGIN_PATH=${CMAKE_BINARY_DIR}/plugins;QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=generic")
    set_target_properties(velora-file-dialog PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/generic")
endif()
